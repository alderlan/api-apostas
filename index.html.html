import { createFileRoute } from "@tanstack/react-router";
import { AuthCard } from "@/components/meu-lucro/AuthCard";
import { Brand } from "@/components/meu-lucro/Brand";
import { Dashboard } from "@/components/meu-lucro/Dashboard";
import { Paywall } from "@/components/meu-lucro/Paywall";
import { useAccess } from "@/hooks/use-access";
import { useSession } from "@/hooks/use-session";
import type { User } from "@supabase/supabase-js";
import { Button } from "@/components/ui/button";
import { supabase } from "@/lib/supabase";

export const Route = createFileRoute("/")({
  head: () => ({
    meta: [
      { title: "Meu Lucro | Gestão de vendas do Mercado Livre" },
      {
        name: "description",
        content:
          "Registre suas vendas do Mercado Livre e veja faturamento, lucro líquido e margem em tempo real.",
      },
      { property: "og:title", content: "Meu Lucro | Gestão de vendas do Mercado Livre" },
      {
        property: "og:description",
        content: "Controle financeiro das suas vendas: faturamento, lucro e margem num só painel.",
      },
      { property: "og:type", content: "website" },
      { name: "twitter:card", content: "summary_large_image" },
    ],
  }),
  component: Index,
});

function Index() {
  const { session, ready } = useSession();
  const status = !ready ? "Conectando..." : session ? "Conectado" : "Desconectado";

  return (
    <main className="mx-auto max-w-6xl px-3 pb-14 pt-5 sm:px-5 sm:pt-7">
      <header className="mb-7 flex items-start justify-between gap-4 sm:items-center">
        <Brand />
        <span className="flex items-center gap-2 text-xs text-muted-foreground">
          <span className={`size-2 rounded-full ${session ? "bg-profit" : "bg-muted-foreground"}`} />
          {status}
        </span>
      </header>

      {ready && (session ? <SignedIn user={session.user} /> : <AuthCard />)}

      <footer className="mt-10 text-center text-xs text-muted-foreground">
        Meu Lucro • Controle financeiro das suas vendas
      </footer>
    </main>
  );
}

function SignedIn({ user }: { user: User }) {
  const access = useAccess(user.id);

  if (access.isPending) {
    return <p className="text-center text-sm text-muted-foreground">Verificando sua assinatura...</p>;
  }
  if (access.error) {
    return (
      <section className="mx-auto max-w-md rounded-2xl border bg-card p-6 text-center shadow-card">
        <h2 className="text-lg font-bold">Não foi possível verificar sua assinatura</h2>
        <p className="mt-2 text-sm text-muted-foreground">{access.error.message}</p>
        <div className="mt-4 flex justify-center gap-2">
          <Button variant="brand" onClick={() => access.refetch()}>
            Tentar de novo
          </Button>
          <Button variant="surface" onClick={() => supabase.auth.signOut()}>
            Sair
          </Button>
        </div>
      </section>
    );
  }
  const { allowed, isAdmin, subscription } = access.data;
  if (!allowed) {
    return (
      <Paywall
        email={user.email ?? ""}
        subscription={subscription}
        onChange={() => void access.refetch()}
        checking={access.isFetching}
      />
    );
  }
  return <Dashboard user={user} isAdmin={isAdmin} paidUntil={subscription?.paid_until ?? null} />;
}
